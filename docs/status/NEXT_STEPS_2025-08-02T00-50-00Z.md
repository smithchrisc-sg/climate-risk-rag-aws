def _sparql_insert_ttl(self, ttl_content: str) -> bool:
    """Insert TTL using SPARQL INSERT with proper prefix handling"""
    try:
        # Separate TTL prefixes from content
        ttl_prefixes, ttl_triples = self._separate_ttl_prefixes(ttl_content)
        
        # Build SPARQL INSERT query with prefixes outside INSERT DATA block
        query = f"""
        {ttl_prefixes}
        INSERT DATA {{
            {ttl_triples}
        }}
        """
        
        return self.kg_manager.execute_sparql_update(query)
    except Exception as e:
        self.logger.error(f"SPARQL INSERT failed: {e}")
        return False

def _separate_ttl_prefixes(self, ttl_content: str) -> tuple[str, str]:
    """Convert TTL prefixes to SPARQL format"""
    lines = ttl_content.split('\n')
    prefix_lines = []
    content_lines = []
    
    for line in lines:
        stripped = line.strip()
        if stripped.startswith('@prefix'):
            # Convert: @prefix kr: <uri> . → PREFIX kr: <uri>
            sparql_prefix = stripped.replace('@prefix', 'PREFIX').rstrip(' .')
            prefix_lines.append(sparql_prefix)
        elif stripped.startswith('@base'):
            sparql_base = stripped.replace('@base', 'BASE').rstrip(' .')
            prefix_lines.append(sparql_base)
        elif stripped and not stripped.startswith('#'):
            content_lines.append(line)
    
    return '\n'.join(prefix_lines), '\n'.join(content_lines)
```

def validate_sparql_injection(self, query: str) -> bool:
    """Enhanced validation that ignores parentheses in string literals"""
    # ... existing dangerous pattern checks ...
    
    # Enhanced parentheses validation
    if not self._check_balanced_parentheses_excluding_literals(query):
        self.logger.warning("Unbalanced parentheses in SPARQL query")
        return False
    
    return True

def _check_balanced_parentheses_excluding_literals(self, query: str) -> bool:
    """Check parentheses balance outside of string literals"""
    paren_count = 0
    in_single_quote = False
    in_double_quote = False
    in_triple_quote = False
    i = 0
    
    while i < len(query):
        char = query[i]
        
        # Handle triple quotes (""")
        if i <= len(query) - 3 and query[i:i+3] == '"""':
            if not in_single_quote and not in_double_quote:
                in_triple_quote = not in_triple_quote
                i += 2
        # Handle double quotes
        elif char == '"' and not in_single_quote and not in_triple_quote:
            if i > 0 and query[i-1] != '\\':  # Not escaped
                in_double_quote = not in_double_quote
        # Handle single quotes  
        elif char == "'" and not in_double_quote and not in_triple_quote:
            if i > 0 and query[i-1] != '\\':  # Not escaped
                in_single_quote = not in_single_quote
        # Count parentheses only outside string literals
        elif not in_single_quote and not in_double_quote and not in_triple_quote:
            if char == '(':
                paren_count += 1
            elif char == ')':
                paren_count -= 1
                if paren_count < 0:  # More closing than opening
                    return False
        
        i += 1
    
    return paren_count == 0  # Should end balanced
```
